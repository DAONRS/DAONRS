# Supabase DB 구축 및 보안 정책 가이드 (v0.0.2)

본 문서는 프로젝트에 필요한 Supabase 테이블 생성, RLS(Row Level Security) 설정 및 Storage 보안 정책 SQL 스크립트입니다. Supabase 대시보드의 **SQL Editor**에 아래 스크립트를 붙여넣고 실행(Run)하십시오.

---

## 1. 테이블 생성 (존재하지 않을 경우에만 생성)

```sql
-- 1-1. 공지사항 (notices) 테이블
CREATE TABLE IF NOT EXISTS notices (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamptz DEFAULT now(),
  title text NOT NULL,
  content text,
  author text DEFAULT '관리자',
  notice_no bigint GENERATED BY DEFAULT AS IDENTITY
);

-- 1-2. 자료실 (archives) 테이블
CREATE TABLE IF NOT EXISTS archives (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamptz DEFAULT now(),
  title text NOT NULL,
  content text,
  author text DEFAULT '관리자',
  file_url text,      
  file_name text,     
  notice_no bigint GENERATED BY DEFAULT AS IDENTITY
);

-- 1-3. 연혁 (history) 테이블
CREATE TABLE IF NOT EXISTS history (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  year text NOT NULL,
  events jsonb NOT NULL, 
  order_index int UNIQUE,
  created_at timestamp with time zone DEFAULT now()
);

-- 1-4. 고객문의 (inquiries) 테이블
CREATE TABLE IF NOT EXISTS public.inquiries (
  id bigint NOT NULL GENERATED BY DEFAULT AS IDENTITY,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  name character varying(100) NOT NULL,
  email character varying(255) NOT NULL,
  company character varying(255) NOT NULL, 
  phone character varying(50) NOT NULL,   
  subject character varying(255) NOT NULL, 
  message text NOT NULL,
  is_read boolean DEFAULT false,
  CONSTRAINT inquiries_pkey PRIMARY KEY (id),
  CONSTRAINT message_min_length CHECK (char_length(message) >= 5)
);

-- 1-5. 스마트 육묘 보급사례 (case_examples) 테이블
CREATE TABLE IF NOT EXISTS case_examples (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  title text NOT NULL,
  content text, 
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
  region TEXT,
  crop TEXT,
  facility_type TEXT,
  area TEXT,
  category TEXT DEFAULT 'example1'
);

-- 1-6. 지식재산권 (intellectual_properties) 테이블
CREATE TABLE IF NOT EXISTS intellectual_properties (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  title text NOT NULL,
  image_url text NOT NULL,
  storage_path text NOT NULL,
  display_order int DEFAULT 0,
  created_at timestamp with time zone DEFAULT now()
);

-- 1-7. 인증서 (certifications) 테이블
CREATE TABLE IF NOT EXISTS certifications (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  title text NOT NULL,
  image_url text NOT NULL,
  storage_path text NOT NULL,
  display_order int DEFAULT 0,
  created_at timestamp with time zone DEFAULT now()
);
```

## 2. RLS 활성화 및 권한 정책(Policy) 설정

```sql
-- ==========================================
-- 2-1. 공지사항 (notices) 테이블 정책
-- ==========================================
ALTER TABLE notices ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "누구나 읽기 가능" ON notices;
CREATE POLICY "누구나 읽기 가능" ON notices FOR SELECT USING (true);

DROP POLICY IF EXISTS "인증된 사용자만 글쓰기 가능" ON notices;
CREATE POLICY "인증된 사용자만 글쓰기 가능" ON notices FOR INSERT TO authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "로그인한 사용자 전체 수정 허용" ON notices;
CREATE POLICY "로그인한 사용자 전체 수정 허용" ON notices FOR UPDATE TO authenticated USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "로그인한 사용자 전체 삭제 허용" ON notices;
CREATE POLICY "로그인한 사용자 전체 삭제 허용" ON notices FOR DELETE TO authenticated USING (true);


-- ==========================================
-- 2-2. 자료실 (archives) 테이블 정책
-- ==========================================
ALTER TABLE archives ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "자료실 누구나 읽기 가능" ON archives;
CREATE POLICY "자료실 누구나 읽기 가능" ON archives FOR SELECT USING (true);

DROP POLICY IF EXISTS "자료실 로그인 사용자 업로드 허용" ON archives;
CREATE POLICY "자료실 로그인 사용자 업로드 허용" ON archives FOR INSERT TO authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "자료실 로그인 사용자 수정 허용" ON archives;
CREATE POLICY "자료실 로그인 사용자 수정 허용" ON archives FOR UPDATE TO authenticated USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "자료실 로그인 사용자 삭제 허용" ON archives;
CREATE POLICY "자료실 로그인 사용자 삭제 허용" ON archives FOR DELETE TO authenticated USING (true);


-- ==========================================
-- 2-3. 고객문의 (inquiries) 테이블 정책
-- ==========================================
ALTER TABLE public.inquiries ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Enable insert for anon users" ON public.inquiries;
CREATE POLICY "Enable insert for anon users" ON public.inquiries FOR INSERT TO anon WITH CHECK (true);

DROP POLICY IF EXISTS "Enable read for authenticated users" ON public.inquiries;
CREATE POLICY "Enable read for authenticated users" ON public.inquiries FOR SELECT TO authenticated USING (true);

DROP POLICY IF EXISTS "Enable update for authenticated users" ON public.inquiries;
CREATE POLICY "Enable update for authenticated users" ON public.inquiries FOR UPDATE TO authenticated USING (true) WITH CHECK (true);


-- ==========================================
-- 2-4. 연혁 (history) 테이블 정책
-- ==========================================
ALTER TABLE history ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "연혁 누구나 조회 가능" ON history;
CREATE POLICY "연혁 누구나 조회 가능" ON history FOR SELECT USING (true);

DROP POLICY IF EXISTS "연혁 관리자 전체 권한" ON history;
CREATE POLICY "연혁 관리자 전체 권한" ON history FOR ALL TO authenticated USING (true) WITH CHECK (true);


-- ==========================================
-- 2-5. 스마트 육묘 보급사례 (case_examples) 테이블 정책
-- ==========================================
ALTER TABLE case_examples ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "사례 누구나 조회 가능" ON case_examples;
CREATE POLICY "사례 누구나 조회 가능" ON case_examples FOR SELECT USING (true);

DROP POLICY IF EXISTS "사례 관리자 전체 권한" ON case_examples;
CREATE POLICY "사례 관리자 전체 권한" ON case_examples FOR ALL TO authenticated USING (true) WITH CHECK (true);


-- ==========================================
-- 2-6. 지식재산권 (intellectual_properties) 테이블 정책
-- ==========================================
ALTER TABLE intellectual_properties ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "지식재산권 누구나 조회 가능" ON intellectual_properties;
CREATE POLICY "지식재산권 누구나 조회 가능" ON intellectual_properties FOR SELECT USING (true);

DROP POLICY IF EXISTS "지식재산권 관리자 전체 권한" ON intellectual_properties;
CREATE POLICY "지식재산권 관리자 전체 권한" ON intellectual_properties FOR ALL TO authenticated USING (true) WITH CHECK (true);


-- ==========================================
-- 2-7. 인증서 (certifications) 테이블 정책
-- ==========================================
ALTER TABLE certifications ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "인증서 누구나 조회 가능" ON certifications;
CREATE POLICY "인증서 누구나 조회 가능" ON certifications FOR SELECT USING (true);

DROP POLICY IF EXISTS "인증서 관리자 전체 권한" ON certifications;
CREATE POLICY "인증서 관리자 전체 권한" ON certifications FOR ALL TO authenticated USING (true) WITH CHECK (true);
```

## 3. 스토리지(Storage) RLS 정책 설정

프로젝트에서 사용하는 스토리지 버킷인 **`daonrs`**와 **`images`** 버킷에 대한 정책을 한 번에 설정합니다.

```sql
-- 기존 스토리지 정책 삭제
DROP POLICY IF EXISTS "스토리지 파일 조회 허용" ON storage.objects;
DROP POLICY IF EXISTS "스토리지 업로드 허용" ON storage.objects;
DROP POLICY IF EXISTS "스토리지 파일 삭제 허용" ON storage.objects;

-- 3-1. 파일 조회 권한: 누구나 daonrs, images 버킷의 파일 조회 가능
CREATE POLICY "스토리지 파일 조회 허용" 
ON storage.objects FOR SELECT 
USING (bucket_id IN ('daonrs', 'images'));

-- 3-2. 파일 업로드 권한: 로그인한 관리자만 daonrs, images 버킷에 업로드 가능
CREATE POLICY "스토리지 업로드 허용" 
ON storage.objects FOR INSERT 
TO authenticated 
WITH CHECK (bucket_id IN ('daonrs', 'images'));

-- 3-3. 파일 삭제 권한: 로그인한 관리자만 daonrs, images 버킷의 파일 삭제 가능
CREATE POLICY "스토리지 파일 삭제 허용" 
ON storage.objects FOR DELETE 
TO authenticated 
USING (bucket_id IN ('daonrs', 'images'));
```
